%=============================================================================
%
%                     BUSINESS LETTER FORMAT MODULE
%
%=============================================================================
%
% References:
%   - SECNAV M-5216.5 (DON Correspondence Manual), Chapter 11
%   - MCO 5216.20B w/Admin CH-4 (Marine Corps Supplement)
%
% Key differences from Standard Naval Letter:
%   - ID symbols in upper LEFT corner (not right)
%   - Date format: Month spelled out (January 5, 2015)
%   - Has salutation (Dear Mr./Ms.) and complimentary close (Sincerely,)
%   - Paragraphs NOT numbered, indented 4 spaces (0.5")
%   - References in the body, enclosure notations after the signature
%   - Inside address 2-8 lines below date
%
%=============================================================================


%=============================================================================
%                         ID SYMBOLS BLOCK (Upper LEFT)
%=============================================================================
% SECNAV Ch 11: ID symbols in upper LEFT corner, blocked one below the other
% Date format: Month spelled out, day in numerals, comma, full year

\newcommand{\printBusinessIdentification}{%
    \noindent
    \begin{tabular}[t]{@{}p{\textwidth}@{}}%
        \optionalLine{\DocumentSSIC}%
        \optionalLine{\DocumentSerial}%
        \BusinessDate%
    \end{tabular}%
}
\newcommand{\printDateAndTitle}{%
    \printBusinessIdentification\par\vspace{12pt}%
}


%=============================================================================
%                    INSIDE ADDRESS AND SALUTATION
%=============================================================================
% SECNAV Ch 11, Para 11-4: Inside address 2-8 lines below date
% SECNAV Ch 11, Para 11-6: Salutation with colon (Dear Mr. Jones:)

\newcommand{\printAddressBlock}{%
    % Inside address block - spacing AFTER each element avoids stacking
    % with the 12pt from \printDateAndTitle
    \ifNotEmptyElse{\BusinessRecipientAddress}{%
        \noindent \BusinessRecipientAddress\par\vspace{12pt}%
    }{%
    \ifNotEmpty{\ToLine}{%
        \noindent
        \optionalLine{\ToLine}%
        \optionalLine{\ToLineTwo}%
        \optionalLine{\ToLineThree}%
        \optionalLine{\ToLineFour}%
        \par\vspace{12pt}%
    }%
    }%
    %
    % Attention line (optional) - SECNAV Ch 11, Para 11-5
    \ifNotEmpty{\AttentionLine}{%
        \noindent Attention: \AttentionLine
        \par\vspace{12pt}%
    }%
    %
    % Salutation with colon - SECNAV Ch 11, Para 11-6
    \noindent \BusinessSalutation
    %
    % Subject line (optional) - SECNAV Ch 11, Para 11-7
    \ifNotEmpty{\SubjectLine}{%
        \par\vspace{12pt}%
        \noindent SUBJECT: \MakeUppercase{\SubjectLine}%
    }%
    %
    % CUI block (if enabled)
    \ifCUIEnabled
        \par\vspace{12pt}%
        \printCUIDesignation%
        \ifNotEmpty{\CUITransmittalNote}{\par\noindent\CUITransmittalNote\par}%
    \fi
}


%=============================================================================
%                         BODY TEXT FORMATTING
%=============================================================================
% SECNAV Ch 11, Para 11-8:
%   - Single-space within paragraphs
%   - Double-space between paragraphs
%   - Main paragraphs indented 4 spaces (0.5")
%   - Subparagraphs indented additional 8 spaces
%   - DO NOT number main paragraphs

\newcommand{\businessLetterParIndent}{0.5in}

% Apply paragraph indentation for business letter body
\newcommand{\setupBusinessLetterBody}{%
    \setlength{\parindent}{\businessLetterParIndent}%
    \setlength{\parskip}{12pt}% Double-space between paragraphs
}


%=============================================================================
%                  SIGNATURE BLOCK (Business Letter Style)
%=============================================================================
% SECNAV Ch 11, Para 11-10-12:
%   - Complimentary close "Sincerely," CENTERED, 2 lines below text
%   - Signature block CENTERED, 4 lines below "Sincerely,"
%   - Name in ALL CAPS
%   - Optional: Rank, Title, "By direction of the Commander"

\newcommand{\printSignature}{%
    % Complimentary close - CENTERED, 2 lines below text
    \par\vspace{24pt}%
    \begin{center}
        \BusinessClose
    \end{center}
    %
    % Signature block - CENTERED, 4 lines below complimentary close
    \vspace{48pt}%
    \begin{center}
        \begin{minipage}{3in}
            \centering
            % Digital signature field (empty box for CAC signing)
        \ifHasDigitalSigField
            \DigitalSignatureBox
        \else
            % Optional: Signature image
            \ifNotEmpty{\SignatureImage}{%
                \IfFileExists{attachments/\SignatureImage}{%
                    \includegraphics[width=1.5in,height=0.5in,keepaspectratio]{attachments/\SignatureImage}\\
                    \vspace{6pt}%
                }{}%
            }%
        \fi
            % Required: Name in ALL CAPS
            \ifNotEmptyElse{\SignatoryAbbrev}{\SignatoryAbbrev\\}{\ifNotEmpty{\SignatoryName}{\MakeUppercase{\SignatoryName}\\}}%
            % Optional: Rank
            \optionalLine{\SignatoryRank}%
            % Optional: Title
            \optionalLine{\SignatoryTitle}%
            % Optional: By direction authority line
            \optionalField{\ByDirection}%
        \end{minipage}
    \end{center}
}


%=============================================================================
%                         CLOSING NOTATIONS
%=============================================================================
% SECNAV M-5216.5 Ch 11, ¶¶10–12 and Figure 11-3
\renewcommand{\printEnclosureList}{}% Business enclosures belong after the signature.
\providecommand{\BusinessEnclosureLabel}{Enclosures}
\providecommand{\BusinessSeparateMailing}{}
\newcounter{businessEnclosure}

\newcommand{\printClosingNotations}{%
    \ifnum\value{enclmax}>0
        \ifdefined\BusinessEnclosureCountOnly
            \par\vspace{12pt}\noindent Enclosures (\arabic{enclmax})\par
        \else
            \dondocsBeginList{\BusinessEnclosureLabel:\hspace{2\fontdimen2\font}}%
            \setcounter{businessEnclosure}{0}%
            \loop\ifnum\value{businessEnclosure}<\value{enclmax}%
                \stepcounter{businessEnclosure}%
                \ifnum\value{enclmax}=1
                    \ifhyperlinks
                        \dondocsEntry{}{\hyperlink{enclosure1}{\csname encl@title@1\endcsname}}%
                    \else
                        \dondocsEntry{}{\csname encl@title@1\endcsname}%
                    \fi
                \else
                    \ifhyperlinks
                        \dondocsEntry{\hyperlink{enclosure\arabic{businessEnclosure}}{\textcolor{blue}{\arabic{businessEnclosure}.}}}{\csname encl@title@\arabic{businessEnclosure}\endcsname}%
                    \else
                        \dondocsEntry{\arabic{businessEnclosure}.}{\csname encl@title@\arabic{businessEnclosure}\endcsname}%
                    \fi
                \fi
            \repeat
            \dondocsEndList
        \fi
    \fi
    \ifdefempty{\BusinessSeparateMailing}{}{%
        \dondocsBeginList{Separate Mailing:\hspace{2\fontdimen2\font}}%
        \dondocsEntry{}{\BusinessSeparateMailing}%
        \dondocsEndList
    }%
}


%=============================================================================
%                              PAGE STYLE
%=============================================================================
% SECNAV Ch 11, Para 11-16:
%   - First page: NOT numbered
%   - Second+ pages: Centered, 1/2 inch from bottom, starting with "2"

\fancypagestyle{firstpage}{%
    \fancyhf{}%
    \fancyhead[C]{\placeClassificationMarkings}%
    \fancyfoot[C]{\smartPageNumber}% No page number on first page
    \renewcommand{\headrulewidth}{0pt}%
    \renewcommand{\footrulewidth}{0pt}%
}

% Follow the left-margin direction in ¶¶1 and 14. Figure 11-3's
% right-hand example conflicts with that written instruction.
\setContinuationLabel{SUBJECT:}
\renewcommand{\dondocsReserveContHead}{%
    \sbox\dondocsContBox{%
        \parbox{\textwidth}{%
            \vspace*{24pt}%
            \applyDocumentFontSize
            \printBusinessIdentification\par
            \printContinuationSubject
        }%
    }%
    \setlength{\dondocsContExtra}{%
        \dimexpr\ht\dondocsContBox+\dp\dondocsContBox-\headheight\relax}%
    \ifdim\dondocsContExtra>0pt
        \addtolength{\headheight}{\dondocsContExtra}%
        \addtolength{\textheight}{-\dondocsContExtra}%
    \fi
}

\fancypagestyle{documentpage}{%
    \fancyhf{}%
    \fancyhead[C]{\placeClassificationMarkings}%
    \fancyhead[L]{\usebox{\dondocsContBox}}%
    \fancyfoot[C]{\smartPageNumber}%
    \renewcommand{\headrulewidth}{0pt}%
    \renewcommand{\footrulewidth}{0pt}%
}
